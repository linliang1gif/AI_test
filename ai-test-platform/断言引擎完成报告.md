```

   ↓

   ├─ 测试通过 → 记录测试用例

```

def rule_valid_phone(data):
    phone = data.get('phone')

### 2. 自定义业务规则

根据实际业务需求,注册自定义规则:

```python
# 在 executor/real_test_executor.py 中添加
def __init__(self):
    # ...
    self.assertion_engine = get_assertion_engine()
    
    # 注册自定义规则
    self._register_custom_rules()

def _register_custom_rules(self):
    """注册自定义业务规则"""
    
    def rule_valid_order_status(data):
        status = data.get('status')
        valid_statuses = ['pending', 'paid', 'shipped', 'completed']
        if status not in valid_statuses:
            return False, f"订单状态无效: {status}"
        return True, f"订单状态有效: {status}"
    
    self.assertion_engine.register_business_rule(
        'valid_order_status', 
        rule_valid_order_status
    )
```
