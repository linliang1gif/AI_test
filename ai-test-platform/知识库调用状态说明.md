### 步骤2: 在Agent Service中集成

修改 `agent/agent_service.py`:

```python
from utils.knowledge_prompt_helper import get_knowledge_helper

class AgentService:
    def __init__(self):
        self.llm_client = get_llm_client()
        self.helper = get_knowledge_helper()  # 新增
    
    def analyze(self, requirement: str) -> dict:
        # 搜索相关API
        apis = self.helper.search_related_apis(requirement, top_k=5)
        
        # 构建增强prompt
        api_context = "\\n".join([
            f"{api['method']} {api['path']} - {api['summary']}"
            for api in apis
        ])
        
        prompt = f\"\"\"
分析需求: {requirement}

系统中相关的API:
{api_context}

请判断是否需要测试。
\"\"\"
        
        response = self.llm_client.generate(prompt)
        return response
```

### 步骤3: 在Case Generator中集成

修改 `case_generator/case_builder.py`:

```python
from utils.knowledge_prompt_helper import get_knowledge_helper

class CaseBuilder:
    def __init__(self):
        self.helper = get_knowledge_helper()  # 新增
    
    def build_cases(self, module_info, scenarios, target_count, priority):
        module_name = module_info['name']
        
        # 搜索相关知识
        apis = self.helper.search_related_apis(module_name, top_k=5)
        backend_code = self.helper.search_related_code(module_name, "backend", top_k=3)
        
        # 构建知识上下文
        knowledge_context = f\"\"\"
相关API ({len(apis)}个):
{self._format_apis(apis)}

相关后端代码 ({len(backend_code)}个):
{self._format_code(backend_code)}
\"\"\"
        
        # 生成用例
        prompt = f\"\"\"
为模块 {module_name} 生成测试用例

{knowledge_context}

场景: {scenarios}
目标数量: {target_count}

请生成详细的测试用例。
\"\"\"
        
        cases = self.llm_client.generate(prompt)
        return cases
```

```python
# agent/llm_client.py

from utils.knowledge_prompt_helper import get_knowledge_helper

class LLMClient:
    def __init__(self):
        self.helper = get_knowledge_helper()
    
    def generate(self, prompt: str, **kwargs) -> str:
        # 自动增强所有prompt
        if "测试" in prompt or "用例" in prompt:
            # 提取关键词
            keywords = self._extract_keywords(prompt)
            # 增强prompt
            prompt = self.helper.enhance_prompt_with_api_knowledge(
                prompt, keywords, top_k=3
            )
        
        # 原有逻辑
        return self._original_generate(prompt, **kwargs)
```
