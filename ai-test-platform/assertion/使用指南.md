# 断言系统使用指南 📖

## 🎯 快速开始

### 1. 最简单的方式 - 使用快捷断言

```python
from assertion import quick_assertions, AssertionValidator

# 创建标准断言(状态码200 + 响应时间<2秒 + 返回码0 + 数据不为空)
configs = quick_assertions().build()

# 准备响应数据
response_data = {
    'status_code': 200,
    'response_time': 0.5,
    'body': {
        'code': 0,
        'data': {'user_id': 123}
    }
}

# 验证
validator = AssertionValidator()
results = validator.validate_all(configs, response_data)

# 查看结果
validator.print_results()
```

## 🔧 进阶使用

### 2. 自定义断言 - 流式API

```python
from assertion import AssertionBuilder, AssertionValidator

# 使用流式API构建断言
builder = AssertionBuilder()
configs = (builder
    # 断言1: 状态码
    .new_assertion("状态码200", "HTTP状态码应为200")
    .equals(200)
    .at_path("status_code")
    .blocker()  # 阻塞级
    
    # 断言2: 响应时间
    .new_assertion("响应时间<1秒")
    .less_than(1.0)
    .at_path("response_time")
    .major()  # 主要级
    
    # 断言3: 业务返回码
    .new_assertion("返回码为0")
    .equals(0)
    .at_path("body.code")
    .critical()  # 严重级
    .with_retry(max_retries=3)  # 失败重试3次
    
    # 断言4: 数据不为空
    .new_assertion("数据不为空")
    .not_empty()
    .at_path("body.data")
    .critical()
    .with_tags("data", "validation")  # 添加标签
    
    # 断言5: 用户ID存在
    .new_assertion("用户ID存在")
    .exists()
    .at_path("body.data.user_id")
    .major()
    
    # 构建
    .build())

# 验证
validator = AssertionValidator()
results = validator.validate_all(configs, response_data)
```

### 5. 知识库集成

```python
from assertion import (
    AssertionBuilder,
    AssertionValidator,
    AssertionKnowledgeIntegration
)
from knowledge.knowledge_manager import get_knowledge_manager

# 创建知识库集成
km = get_knowledge_manager()
integration = AssertionKnowledgeIntegration(km)

# 创建和验证断言
configs = quick_assertions().build()
validator = AssertionValidator()
results = validator.validate_all(configs, response_data)

# 记录到知识库
for config, result in zip(configs, results):
    # 记录结果
    record_id = integration.record_assertion_result(
        config, result,
        context={
            'test_case_id': 'TC001',
            'api_endpoint': '/api/test',
            'request_data': {...},
            'response_data': response_data
        }
    )
    
    # 如果失败,重点记录
    if not result.passed:
        failure_id = integration.record_assertion_failure(
            config, result,
            context={'test_case_id': 'TC001'}
        )

# 获取统计
stats = integration.get_assertion_statistics()
print(f"总计: {stats['total']}")
print(f"通过率: {stats['pass_rate']:.1%}")

# 获取改进建议
for config in configs:
    suggestions = integration.suggest_assertion_improvements(config)
    if suggestions:
        print(f"\n{config.name} 的改进建议:")
        for suggestion in suggestions:
            print(f"  - {suggestion}")

# 查找相似失败
similar = integration.find_similar_assertion_failures(config, limit=5)

# 导出知识
integration.export_assertion_knowledge("assertion_knowledge.json")
```

## 📚 所有断言操作符

### 比较操作符

```python
.equals(value)              # 等于
.not_equals(value)          # 不等于
.greater_than(value)        # 大于
.less_than(value)           # 小于
.greater_equal(value)       # 大于等于
.less_equal(value)          # 小于等于
```

### 字符串操作符

```python
.contains(text)             # 包含
.not_contains(text)         # 不包含
.starts_with(text)          # 开始于
.ends_with(text)            # 结束于
.matches(pattern)           # 正则匹配
```

### 列表操作符

```python
.in_list([1, 2, 3])        # 在列表中
.not_in([1, 2, 3])         # 不在列表中
```

### 存在性操作符

```python
.exists()                   # 存在
.not_exists()               # 不存在
.not_empty()                # 不为空
.is_empty()                 # 为空
```

### 类型操作符

```python
.is_type("string")          # 类型检查
.is_type("number")
.is_type("boolean")
.is_type("dict")
.is_type("list")
```

### 长度操作符

```python
.length_equals(10)          # 长度等于
```

## 🎨 严重程度设置

```python
.blocker()      # 阻塞级 - 必须通过
.critical()     # 严重级 - 核心功能
.major()        # 主要级 - 重要功能(默认)
.minor()        # 次要级 - 一般功能
.trivial()      # 轻微级 - 可选功能

# 或使用
.with_severity(AssertionSeverity.BLOCKER)
```

## 🔍 完整示例

### 示例1: API测试完整流程

```python
from assertion import (
    AssertionBuilder,
    AssertionValidator,
    AssertionKnowledgeIntegration
)
import requests

# 1. 构建断言
builder = AssertionBuilder()
configs = (builder
    .status_code_200()
    .response_time_under(2.0)
    .business_code_success()
    .data_not_empty()
    .new_assertion("返回token")
    .exists()
    .at_path("body.data.token")
    .critical()
    .build())

# 2. 发送请求
response = requests.post('/api/login', json={
    'username': 'test',
    'password': '123456'
})

# 3. 准备响应数据
response_data = {
    'status_code': response.status_code,
    'response_time': response.elapsed.total_seconds(),
    'headers': dict(response.headers),
    'body': response.json()
}

# 4. 验证断言
validator = AssertionValidator()
results = validator.validate_all(configs, response_data)

# 5. 打印结果
validator.print_results()

# 6. 记录到知识库
integration = AssertionKnowledgeIntegration(km)
for config, result in zip(configs, results):
    integration.record_assertion_result(
        config, result,
        context={
            'api': '/api/login',
            'method': 'POST',
            'response': response_data
        }
    )

# 7. 检查是否全部通过
summary = validator.get_summary()
if summary['all_passed']:
    print("✅ 所有断言通过!")
else:
    print(f"❌ {summary['failed']} 个断言失败")
```

### 示例2: 批量API测试

```python
from assertion import quick_assertions, AssertionValidator

# 测试多个API
apis = [
    {'url': '/api/users', 'method': 'GET'},
    {'url': '/api/orders', 'method': 'GET'},
    {'url': '/api/products', 'method': 'GET'}
]

# 使用相同的断言配置
configs = quick_assertions().build()
validator = AssertionValidator()

for api in apis:
    print(f"\n测试 {api['url']}...")
    
    # 发送请求
    response = requests.get(api['url'])
    response_data = {
        'status_code': response.status_code,
        'response_time': response.elapsed.total_seconds(),
        'body': response.json()
    }
    
    # 验证
    validator.clear_results()
    results = validator.validate_all(configs, response_data)
    
    # 简单输出
    summary = validator.get_summary()
    print(f"  通过率: {summary['pass_rate']:.1%}")
```

## 🐛 常见问题

### Q1: JSON路径如何写?

```python
# 简单路径
.at_path("code")                    # 根级字段
.at_path("data.user_id")           # 嵌套字段
.at_path("data.profile.age")       # 多层嵌套

# 数组访问
.at_path("data.0.id")              # 第一个元素的id
.at_path("data.1.name")            # 第二个元素的name
```

### Q2: 如何验证数组长度?

```python
.new_assertion("数组长度为10")
.length_equals(10)
.at_path("body.data")
```

### Q3: 如何验证多个可能的值?

```python
.new_assertion("状态码为2xx")
.in_list([200, 201, 204])
.at_path("status_code")
```

### Q4: 如何自定义业务规则?

```python
from assertion import get_assertion_engine

engine = get_assertion_engine()

# 注册自定义规则
def rule_valid_phone(data):
    phone = data.get('phone')
    if not phone:
        return False, "phone不存在"
    if not re.match(r'^1[3-9]\d{9}$', phone):
        return False, f"phone格式无效: {phone}"
    return True, f"phone格式有效: {phone}"

engine.register_business_rule('valid_phone', rule_valid_phone)

# 使用
engine.assert_business_rule('valid_phone', {'phone': '13800138000'})
```
