### 生成的脚本示例
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
用户认证功能自动化测试脚本
Generated by AI Test Platform
Framework: pytest
"""

import pytest
import requests
import json
from datetime import datetime

class TestUserAuthentication:
    """用户认证功能自动化测试脚本"""
    
    def setup_method(self):
        """测试前置条件"""
        self.base_url = "http://localhost:8080"
        self.headers = {"Content-Type": "application/json"}
        
    def test_api_health_check(self):
        """API健康检查测试"""
        response = requests.get(f"{self.base_url}/health", timeout=5)
        assert response.status_code == 200
        print(f"✅ API健康检查通过: {response.status_code}")
    
    def test_user_authentication(self):
        """用户认证测试"""
        login_data = {
            "username": "test_user",
            "password": "test_password"
        }
        
        response = requests.post(
            f"{self.base_url}/api/auth/login",
            json=login_data,
            headers=self.headers,
            timeout=10
        )
        
        assert response.status_code in [200, 201]
        response_data = response.json()
        assert "token" in response_data or "access_token" in response_data
        print(f"✅ 用户认证测试通过")
    
    def teardown_method(self):
        """测试后清理"""
        print(f"🧹 测试清理完成 - {datetime.now()}")

if __name__ == "__main__":
    pytest.main([__file__, "-v", "--tb=short"])
```

```bash
curl http://127.0.0.1:8081/api/automation/scripts
```

```bash
curl -X POST http://127.0.0.1:8081/api/ai/generate \
  -H "Content-Type: application/json" \
  -d '{
    "type": "automation_script",
    "framework": "pytest",
    "language": "Python",
    "test_type": "API Tests"
  }'
```

### 3. 从测试用例生成
```bash
curl -X POST http://127.0.0.1:8081/api/automation/scripts/generate \
  -H "Content-Type: application/json" \

    "framework": "pytest",
    "language": "Python"
  }'
```

### 4. 下载脚本
```bash
curl http://127.0.0.1:8081/api/automation/scripts/1/download \
  -o test_script.py
```

