## 🚀 使用示例

### 1. 智能生成单个字段
```python
from test_data.data_factory import factory

# 智能生成邮箱(基于用户名)
email = factory.smart_generate('user_email', {'username': 'john'})
# 结果: john@example.com

# 智能生成手机号(基于国家)
phone = factory.smart_generate('contact_phone', {'country': 'CN'})
# 结果: 13800138000

# 智能生成价格(基于类别)
price = factory.smart_generate('product_price', {'category': '电子产品'})
# 结果: 2999.99
```

### 2. 智能生成对象
```python
# 定义schema
user_schema = {
    'user_id': 'string',
    'username': 'string',
    'email': 'string',
    'phone': 'string',
    'age': 'integer',
    'status': 'string'
}

# 智能生成用户对象
user = factory.smart_object(user_schema, {
    'role': 'adult',
    'country': 'CN'
})
```

### 3. 字段分析
```python
# 分析字段
analysis = factory.analyze_field('user_email')
print(analysis['field_type'])      # personal
print(analysis['data_type'])       # string
print(analysis['suggestions'])     # ['验证邮箱格式', ...]
print(analysis['constraints'])     # {'format': 'email', ...}
```

### 4. Schema分析
```python
# 分析schema
schema = {
    'order_id': 'string',
    'user_id': 'string',
    'total_amount': 'float',
    'status': 'string'
}

analysis = factory.analyze_schema(schema)
print(analysis['entity_type'])     # order
print(analysis['relationships'])   # [{'entity': 'user', ...}]
print(analysis['recommendations']) # ['建议添加时间戳字段']
```

### 5. 测试场景建议
```python
# 获取测试场景建议
scenarios = factory.suggest_scenarios('user')
for scenario in scenarios:
    print(f"{scenario['name']} - {scenario['type']}")
# 输出:
# 正常注册 - positive
# 重复邮箱注册 - negative
# 无效邮箱格式 - negative
```

### 6. 数据质量评估
```python
# 评估单个数据
data = {
    'user_id': 'USR12345',
    'email': 'test@example.com',
    'phone': '13800138000',
    'age': 25
}

result = factory.evaluate_quality(data)
print(f"总分: {result['total_score']}")
print(f"等级: {result['level']}")
print(f"问题: {result['issues']}")
print(f"建议: {result['suggestions']}")
```

### 7. 批量质量评估
```python
# 批量评估
data_list = [factory.user() for _ in range(10)]
result = factory.batch_evaluate(data_list)

print(f"平均分: {result['average_score']}")
print(f"质量分布: {result['quality_distribution']}")
```

### 8. 生成统计

stats = factory.generation_stats()
