**实现细节**:
- 后端API: `POST /api/testcases/{testcase_id}/execute`

## 🎯 核心代码

### 后端API (backend_api_server.py)

```python
@app.post("/api/testcases/{testcase_id}/generate-script")
async def generate_test_script(testcase_id: int):
    """为测试用例生成自动化脚本"""
    testcase = next((tc for tc in test_cases_db if tc['id'] == testcase_id), None)
    if not testcase:
        raise HTTPException(status_code=404, detail="测试用例不存在")
    
    script_content = _generate_pytest_script(testcase)
    
    return {
        "success": True,
        "script": script_content,
        "message": "脚本生成成功"
    }

@app.post("/api/testcases/{testcase_id}/execute")
async def execute_test_case(testcase_id: int):
    """执行测试用例"""
    testcase = next((tc for tc in test_cases_db if tc['id'] == testcase_id), None)
    if not testcase:
        raise HTTPException(status_code=404, detail="测试用例不存在")
    
    from executor.real_test_executor import get_test_executor
    executor = get_test_executor()
    
    result = await executor.execute_test_case(testcase)
    
    # 更新测试用例状态
    testcase['status'] = result['status']
    testcase['lastRun'] = time.strftime("%Y-%m-%d %H:%M:%S")
    
    # 记录执行历史
    test_run = {
        "id": len(test_runs_db) + 1,
        "testcase_id": testcase_id,
        "testcase_title": testcase['title'],
        "status": result['status'],
        "duration": result.get('response_time', '0ms'),
        "executed_at": time.strftime("%Y-%m-%d %H:%M:%S"),
        "details": result.get('details', ''),
        "message": result.get('message', ''),
        "assertions": result.get('assertions', {})
    }
    test_runs_db.append(test_run)
    
    return {
        "success": True,
        "result": result,
        "message": "测试执行完成"
    }
```

async def execute_mock_test(self, test_case: Dict[str, Any]) -> Dict[str, Any]:
    """模拟执行测试用例(用于演示和开发)"""
    start_time = time.time()
    
    title = test_case.get('title', '')
    steps = test_case.get('steps', [])
    priority = test_case.get('priority', 'medium')
    
    # 模拟执行时间
    if priority == 'high':
        await asyncio.sleep(0.5)
    elif priority == 'medium':
        await asyncio.sleep(0.3)
    else:
        await asyncio.sleep(0.2)
    
    # 模拟执行步骤
    print(f"🧪 模拟执行: {title}")
    for i, step in enumerate(steps, 1):
        print(f"   步骤{i}: {step}")
    
    # 90%的概率通过
    import random
    passed = random.random() > 0.1
    
    response_time_ms = (time.time() - start_time) * 1000
    
    if passed:
        return {
            "status": "passed",
            "message": f"测试通过: {title}",
            "details": f"成功执行 {len(steps)} 个步骤",
            "response_time": f"{response_time_ms:.2f}ms",
            "assertions": {
                "total": len(steps),
                "passed": len(steps),
                "failed": 0
            }
        }
    else:
        failed_step = random.randint(1, len(steps))
        return {
            "status": "failed",
            "message": f"测试失败: 步骤{failed_step}执行失败",
            "details": f"在步骤{failed_step}处失败",
            "response_time": f"{response_time_ms:.2f}ms",
            "assertions": {
                "total": len(steps),
                "passed": failed_step - 1,
                "failed": 1
            }
        }
