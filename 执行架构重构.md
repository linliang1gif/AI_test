```python
from core import (
    ExecutionResult,
    HealingRecord,
    HealingLevel,
    TestCaseStatus
)
from datetime import datetime

def heal(self, result: ExecutionResult) -> HealingRecord:
    # 分析错误类型
    error_type = self._analyze_error(result)
    healing_level = self._determine_level(error_type)
    
    # 应用修复策略
    success = self._apply_healing(result, healing_level)
    
    return HealingRecord(
        test_case_id=result.test_case_id,
        error_type=error_type,
        healing_level=healing_level,
        healing_strategy=self._get_strategy(healing_level),
        success=success,
        timestamp=datetime.now(),
        details=f"Applied {healing_level.value} healing"
    )
```

### 2.4 更新 modules/report/report_generator.py

**修改示例**:

```python
from core import (
    ExecutionResult,
    TestReport,
    TestCaseStatus
)
from datetime import datetime

def generate(self, results: List[ExecutionResult], format: str = "json") -> TestReport:
    start_time = min(r.start_time for r in results)
    end_time = max(r.end_time for r in results)
    
    passed = sum(1 for r in results if r.status == TestCaseStatus.PASSED)
    failed = sum(1 for r in results if r.status == TestCaseStatus.FAILED)
    
    return TestReport(
        report_id=f"REPORT_{datetime.now().strftime('%Y%m%d%H%M%S')}",
        title="Test Execution Report",
        start_time=start_time,
        end_time=end_time,
        duration=(end_time - start_time).total_seconds(),
        total_tests=len(results),
        passed=passed,
        failed=failed,
        skipped=0,
        pass_rate=passed / len(results) * 100 if results else 0,
        results=results,
        healing_summary=self._calculate_healing_summary(results),
        healed_cases=[r.test_case_id for r in results if r.healing_applied]
    )
```

```python
from core import DataType

def generate_data(self, schema: Dict[str, Any], data_type: str) -> Dict[str, Any]:
    # 转换字符串为枚举
    dt = DataType(data_type)
    
    if dt == DataType.VALID:
        return self._generate_valid_data(schema)
    elif dt == DataType.BOUNDARY:
        return self._generate_boundary_data(schema)
    elif dt == DataType.INVALID:
        return self._generate_invalid_data(schema)
    else:
        return {}
```

```python
async def generate_testcases(file: UploadFile = File(...)):

                "data_type": tc.data_type.value,
```

**测试执行**:

```python
@app.post("/api/testcases/{testcase_id}/execute")
async def execute_testcase(testcase_id: int, config: Dict[str, Any]):
    """执行测试用例 - 调用 modules.executor"""
    
    # 从数据库获取测试用例
    test_case_data = get_testcase_from_db(testcase_id)
    
    # 🔧 转换为 TestCase 对象
    test_case = create_test_case(**test_case_data)
    
    # 🔧 使用 modules SDK
    engine = ExecutionEngine(config)
    result = engine.execute(test_case)
    
    # 🔧 如果失败，尝试修复
    if result.status == TestCaseStatus.FAILED:
        healing_engine = HealingEngine()
        healing_record = healing_engine.heal(result)
        
        if healing_record.success:
            result = engine.execute(test_case)
            result.healing_applied = True
            result.healing_level = healing_record.healing_level
    
    return {
        "success": True,
        "result": {
            "status": result.status.value,
            "duration": result.duration,
            "healing_applied": result.healing_applied
        }
    }
```
